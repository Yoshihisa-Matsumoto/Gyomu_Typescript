# node_modules をすべて削除
Get-ChildItem -Path . -Directory -Filter node_modules -Recurse -Force |
    Remove-Item -Recurse -Force

# pnpm のロックファイルを削除（必要なら）
Remove-Item pnpm-lock.yaml -Force -ErrorAction SilentlyContinue

# Turbo のキャッシュ・ログを削除
Get-ChildItem -Path . -Directory -Filter .turbo -Recurse -Force |
    Remove-Item -Recurse -Force

# TypeScript の増分ビルド情報を削除
Get-ChildItem -Path . -File -Filter *.tsbuildinfo -Recurse -Force |
    Remove-Item -Force

# dist を削除
Get-ChildItem -Path . -Directory -Filter dist -Recurse -Force |
    Remove-Item -Recurse -Force
